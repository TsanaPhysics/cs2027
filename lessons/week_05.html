# =====================================================================
# โครงการ: ตัวแบบจำลองการคำนวณฟิสิกส์และเคมีด้วยตัวแปรและเงื่อนไข
# ผู้พัฒนา: ผู้ช่วยศาสตราจารย์ ดร.ชีวะ ทัศนา
# สังกัด: คณะวิทยาศาสตร์และเทคโนโลยี มหาวิทยาลัยราชภัฏรำไพพรรณี
# =====================================================================

from typing import Dict, Any

# ---------------------------------------------------------------------
# 1. NEWTON'S 2ND LAW SIMULATOR (F = ma with Safety Condition)
# ---------------------------------------------------------------------
def calculate_newton_force(mass_kg: float, acc_ms2: float, safety_limit_newtons: float = 100.0) -> Dict[str, Any]:
    """
    คำนวณแรงลัพธ์ตามกฎข้อที่ 2 ของนิวตัน F = ma และตรวจสอบพิกัดความปลอดภัย
    mass_kg: มวลของวัตถุ (กิโลกรัม)
    acc_ms2: ความเร่ง (เมตร/วินาทีกำลังสอง)
    """
    if mass_kg <= 0:
        raise ValueError("มวลของวัตถุต้องมีค่ามากกว่าศูนย์เสมอ")
        
    force = mass_kg * acc_ms2
    is_safe = force <= safety_limit_newtons
    
    status_msg = "แรงอยู่ในเกณฑ์ปลอดภัยสำหรับการทดลอง" if is_safe else "คำเตือน: แรงเกินพิกัดความปลอดภัยของโครงสร้าง!"
    
    return {
        "mass": mass_kg,
        "acceleration": acc_ms2,
        "force_newtons": round(force, 2),
        "is_safe": is_safe,
        "message": status_msg
    }


# ---------------------------------------------------------------------
# 2. CHEMICAL pH CLASSIFIER (Multi-branching if-elif-else)
# ---------------------------------------------------------------------
def classify_chemical_ph(ph_val: float) -> Dict[str, Any]:
    """
    จำแนกระดับความเป็นกรด-ด่างของสารละลาย และคำนวณความเข้มข้นไฮโดรเนียมไอออน
    """
    if ph_val < 0.0 or ph_val > 14.0:
        raise ValueError("ค่า pH มาตรฐานในสารละลายน้ำต้องอยู่ระหว่าง 0.0 ถึง 14.0")
        
    # Multi-branching classification
    if ph_val < 3.0:
        category = "กรดแก่ (Strong Acid)"
        precaution = "อันตรายสูง: สวมถุงมือไนไตรล์และแว่นตานิรภัย"
    elif ph_val < 7.0:
        category = "กรดอ่อน (Weak Acid)"
        precaution = "ความปลอดภัยปานกลาง: ระวังการสัมผัสดวงตา"
    elif ph_val == 7.0:
        category = "เป็นกลาง (Neutral)"
        precaution = "ปลอดภัย: สภาวะสมดุลทางเคมี"
    elif ph_val <= 11.0:
        category = "เบสอ่อน (Weak Base)"
        precaution = "ความปลอดภัยปานกลาง: สารมีความลื่น ล้างด้วยน้ำสะอาด"
    else:
        category = "เบสแก่ (Strong Base)"
        precaution = "อันตรายสูง: มีฤทธิ์กัดกร่อนผิวหนังรุนแรง"
        
    h_ion_conc = 10.0 ** (-ph_val)  # [H3O+] = 10^(-pH)
    
    return {
        "pH": ph_val,
        "category": category,
        "h_concentration_molar": f"{h_ion_conc:.2e} M",
        "safety_precaution": precaution
    }


# ---------------------------------------------------------------------
# EXECUTION & DEMONSTRATION
# ---------------------------------------------------------------------
if __name__ == "__main__":
    print("=== [1] ทดสอบการคำนวณกฎข้อที่ 2 ของนิวตัน (F = ma) ===")
    test_physics = calculate_newton_force(mass_kg=12.5, acc_ms2=9.8)
    print(f"มวล: {test_physics['mass']} kg, ความเร่ง: {test_physics['acceleration']} m/s²")
    print(f"แรงลัพธ์คำนวณได้: {test_physics['force_newtons']} N")
    print(f"สถานะความปลอดภัย: {test_physics['message']}\n")

    print("=== [2] ทดสอบการจำแนกสารละลายตามค่า pH สเกล ===")
    samples = [1.5, 4.2, 7.0, 9.5, 13.0]
    for sample_ph in samples:
        res = classify_chemical_ph(sample_ph)
        print(f"pH {res['pH']:<4} -> {res['category']:<24} | ความเข้มข้น H+: {res['h_concentration_molar']}")

    นักศึกษาเปิดคอมพิวเตอร์พิมพ์โค้ดตามอาจารย์ทีละบรรทัด สร้างโปรแกรมคำนวณดัชนีมวลกาย (BMI) พร้อมเงื่อนไขจำแนกระดับสุขภาพ โดยสังเกตผลกระทบเมื่อลืมใส่ฟังก์ชัน float() ในการแปลงค่า input()

  4. ขั้นสรุปเชื่อมโยง (Elaboration - 15 นาที):

    อภิปรายเปรียบเทียบข้อดี-ข้อจำกัดระหว่าง Scratch (มองเห็นภาพรวมง่าย) กับ Python (คำนวณวิทยาศาสตร์และประมวลผลข้อมูลขนาดใหญ่ได้ดีกว่า)

ชั่วโมงที่ 3-4: ปฏิบัติการ Pair Programming "นักวิทยาศาสตร์จับคู่โปรแกรมเมอร์" 120 นาที
  1. กิจกรรมจับคู่เขียนโค้ด (Pair Programming Workshop - 70 นาที):

    จับคู่นักศึกษา 2 คน สวมบทบาทสลับกันระหว่าง "Driver" (ผู้พิมพ์โค้ด) และ "Navigator" (ผู้ตรวจสอบสูตรและตรรกะ) เลือกแก้โจทย์วิทยาศาสตร์ 1 ข้อ:

    • โจทย์ A (ฟิสิกส์): โปรแกรมคำนวณพลังงานจลน์ \(E_k = \frac{1}{2}mv^2\) และตรวจสอบว่าถ้าความเร็ว \(v > 120\text{ km/h}\) ให้แจ้งเตือนความเร็วอันตราย
    • โจทย์ B (เคมี): โปรแกรมคำนวณความเข้มข้นสารละลาย \(C = n/V\) พร้อมตรวจสอบว่าสารละลายอิ่มตัวหรือไม่
    • โจทย์ C (ชีววิทยา): โปรแกรมตรวจสอบอุณหภูมิร่างกายและชีพจร (Nested If-Else) เพื่อคัดกรองภาวะติดเชื้อเบื้องต้น
  2. กิจกรรม "Live Bug Hunt & Peer Code Review" (35 นาที):

    สุ่มตัวแทน 2-3 คู่ขึ้นแสดงผลงานบนจอโปรเจกเตอร์ อาจารย์และเพื่อนร่วมชั้นช่วยกันตรวจจับบั๊กที่พบบ่อย (เช่น ลืมใส่เครื่องหมาย : หลังคำสั่ง if, ใช้เครื่องหมาย = แทน == ในการเปรียบเทียบ)

  3. สรุปผลและมอบหมายงาน (Debrief & Homework - 15 นาที):

    อาจารย์สรุปภาพรวมและมอบหมายให้นักศึกษาปรับปรุงโค้ดส่งผ่าน Google Classroom เพื่อเตรียมพร้อมสู่การเขียนลูปวนซ้ำ (Loops) ในสัปดาห์ที่ 6

เกณฑ์การวัดและประเมินผลการเรียนรู้ (Authentic Rubrics Matrix)

เกณฑ์การประเมินผลงานการเขียนโปรแกรมจำลองวิทยาศาสตร์ 4 ระดับคุณภาพ พัฒนาตามเกณฑ์สมรรถนะการเขียนโปรแกรมเพื่อการสอนวิทยาศาสตร์ (Sungur & Ozkan, 2026; Listiaji et al., 2026):

มิติการประเมิน ดีมาก (4 คะแนน) ดี (3 คะแนน) พอใช้ (2 คะแนน) ปรับปรุง (1 คะแนน)
1. การจัดการตัวแปรและชนิดข้อมูล (Variable & Data Typing) ตั้งชื่อตัวแปรสื่อความหมายตามสูตรวิทยาศาสตร์ มีการใช้ Type Casting (float, int) ครบถ้วน ถูกต้อง 100% ไม่มีข้อผิดพลาด Type Mismatch ตั้งชื่อตัวแปรได้ถูกต้อง มีการแปลงชนิดข้อมูลครบถ้วน แต่อาจมีชื่อตัวแปรบางตัวที่สั้นหรือไม่ชัดเจน ลืมแปลงชนิดข้อมูลในบางจุด ส่งผลให้การคำนวณคลาดเคลื่อน หรือใช้ชนิดข้อมูลไม่เหมาะสมกับปริมาณทางกายภาพ ไม่เข้าใจแนวคิดเรื่องชนิดข้อมูล ไม่มีการแปลงค่า input() ทำให้โปรแกรมเกิด Error
2. ความถูกต้องของตรรกะเงื่อนไข (Conditional Logic) ใช้ if-elif-else และตัวดำเนินการตรรกะ (and, or) ครอบคลุมทุกช่วงค่าทางวิทยาศาสตร์ ไม่มีช่องว่างเงื่อนไข (No Gap) และไม่มีเงื่อนไขที่ซ้ำซ้อน โครงสร้างเงื่อนไขถูกต้องเป็นส่วนใหญ่ ครอบคลุมกรณีทดสอบหลักเกือบทั้งหมด เงื่อนไขมีช่องโหว่ เช่น ลืมกรณีค่าขอบเขต (Boundary Conditions เช่น กรณีค่าเท่ากับพอดี) หรือใช้เครื่องหมายเปรียบเทียบสลับกัน ตรรกะเงื่อนไขผิดพลาด ไม่สามารถตัดสินใจตามเงื่อนไขที่โจทย์กำหนดได้
3. การถ่ายโอนความรู้คู่ขนาน (Block-to-Text Transfer) สามารถอธิบายความเชื่อมโยงระหว่างบล็อก Scratch กับไวยากรณ์ Python ได้อย่างชัดเจน เข้าใจกลไกการทำงานของทั้งสองกระบวนทัศน์อย่างลึกซึ้ง เปรียบเทียบความเหมือนและความต่างระหว่าง Scratch และ Python ได้ถูกต้องเป็นส่วนใหญ่ เขียนได้เฉพาะบล็อก Scratch หรือเฉพาะ Python อย่างใดอย่างหนึ่ง แต่ไม่สามารถเชื่อมโยงทั้งสองรูปแบบเข้าด้วยกันได้ ไม่เข้าใจความเชื่อมโยงระหว่างการเขียนโปรแกรมแบบบล็อกและแบบข้อความ
4. ความถูกต้องทางวิทยาศาสตร์ (Scientific Accuracy) สูตรการคำนวณทางวิทยาศาสตร์ถูกต้องสมบูรณ์ หน่วยทางกายภาพถูกต้อง มีระบบตรวจสอบข้อมูลนำเข้าป้องกันค่าที่ไม่สมเหตุสมผล (เช่น มวลติดลบ) สูตรคำนวณถูกต้อง ผลลัพธ์ทางวิทยาศาสตร์ถูกต้อง แต่อาจขาดการตรวจสอบข้อมูลนำเข้าที่ผิดพลาด สูตรคำนวณทางวิทยาศาสตร์มีข้อผิดพลาดบางจุด เช่น ลืมคูณสัมประสิทธิ์ หรือหน่วยไม่สอดคล้องกัน สูตรวิทยาศาสตร์ผิดพลาดอย่างสิ้นเชิง ไม่สะท้อนหลักการทางฟิสิกส์หรือเคมี

ข้อสอบวัดสมรรถนะการคิดขั้นสูง (Higher-Order Thinking Skills: HOTS)

ข้อสอบวัดสมรรถนะคัดสรรจาก คลังข้อสอบมาตรฐานวิทยาการคำนวณ สสวท. (60 ข้อ) หมวดการเขียนโปรแกรมแก้ปัญหาวิทยาศาสตร์:

ข้อที่ 1: การวิเคราะห์ข้อผิดพลาด Type Error ในการรับค่าการทดลอง (HOTS: Analyzing & Debugging)

สสวท. หมวด Data Types

สถานการณ์: นักเรียนเขียนโปรแกรมภาษา Python เพื่อคำนวณความเร็วของรถทดลองฟิสิกส์ตามสมการ \(v = s / t\) ดังนี้:

s = input("ระยะทาง (m): ")
t = input("เวลา (s): ")
v = s / t
print("ความเร็วคือ:", v)

เมื่อผู้ใช้ป้อนระยะทาง 100 และเวลา 5 โปรแกรมจะแสดงผลลัพธ์อย่างไร?

ก. แสดงผลลัพธ์: ความเร็วคือ: 20.0
ข. แสดงผลลัพธ์: ความเร็วคือ: 100/5
ค. เกิดข้อผิดพลาด TypeError: unsupported operand type(s) for /: 'str' and 'str' เพราะคำสั่ง input() คืนค่าเป็น String ซึ่งไม่สามารถนำมาหารกันทางคณิตศาสตร์ได้
ง. เกิดข้อผิดพลาด ZeroDivisionError เพราะคอมพิวเตอร์มองไม่เห็นตัวเลข
เฉลย: ค. เกิดข้อผิดพลาด TypeError
บทวิเคราะห์เชิงวิทยาการคำนวณ: ในภาษา Python ฟังก์ชัน input() จะรับข้อมูลเข้ามาเป็นชนิดข้อความ (String) เสมอ แม้ผู้ใช้จะพิมพ์ตัวเลข 100 และ 5 เข้าไป ตัวแปร \(s\) จะมีค่าเป็น "100" และ \(t\) จะมีค่าเป็น "5" ซึ่งตัวดำเนินการหาร / ไม่รองรับข้อมูลชนิดสตริง การแก้ไขที่ถูกต้องคือต้องทำ Type Casting: s = float(input(...)) และ t = float(input(...)) ก่อนนำไปคำนวณ

ข้อที่ 2: การออกแบบเงื่อนไขตรรกศาสตร์จำแนกสถานะของสาร (HOTS: Evaluating & Logic Design)

สสวท. หมวด Conditionals

สถานการณ์: ครูต้องการเขียนโปรแกรมตรวจสอบสถานะของน้ำบริสุทธิ์ ณ ความดันบรรยากาศ 1 atm โดยพิจารณาจากอุณหภูมิ \(T\) (°C):

  • ถ้า \(T \le 0\) เป็น "ของแข็ง (น้ำแข็ง)"
  • ถ้า \(0 < T < 100\) เป็น "ของเหลว (น้ำ)"
  • ถ้า \(T \ge 100\) เป็น "แก๊ส (ไอน้ำ)"

ข้อใดเป็นโครงสร้างโค้ดภาษา Python ที่ถูกต้อง รัดกุม และครอบคลุมทุกช่วงอุณหภูมิทางกายภาพมากที่สุด?

ก.
if T <= 0: print("Solid")
if T > 0 and T < 100: print("Liquid")
if T >= 100: print("Gas")
ข. (โครงสร้าง If-Elif-Else ที่มีประสิทธิภาพสูงสุด)
if T <= 0:
    print("Solid")
elif T < 100:
    print("Liquid")
else:
    print("Gas")
ค.
if T == 0: print("Solid")
elif T == 100: print("Gas")
else: print("Liquid")
ง.
if T >= 100: print("Gas")
else: print("Liquid")
เฉลย: ข. โครงสร้าง If-Elif-Else ที่มีประสิทธิภาพสูงสุด
บทวิเคราะห์เชิงตรรกศาสตร์:
1. ตัวเลือก ข. ใช้โครงสร้าง if-elif-else แบบคัดกรองเป็นลำดับขั้น เมื่อเงื่อนไขแรก T <= 0 ไม่เป็นจริง แสดงว่าค่า \(T > 0\) เสมอโดยอัตโนมัติ ดังนั้นในบล็อก elif T < 100: จึงไม่จำเป็นต้องเขียนเงื่อนไขซ้ำซ้อน T > 0 and ... ช่วยลดภาระการประมวลผลของ CPU
2. และหากไม่เข้าทั้งสองเงื่อนไข บล็อก else: จะรองรับค่า \(T \ge 100\) ทุกค่าอย่างสมบูรณ์แบบโดยไม่มีช่องว่างเงื่อนไข (No Boundary Gap)
